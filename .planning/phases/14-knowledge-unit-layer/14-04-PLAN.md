```powershell
$inventory = Resolve-Path 'integration/analysis/ai_context/knowledge_unit_inventory.json'
$preflightPath = 'integration/analysis/ai_context/knowledge_unit_full_run_preflight.json'
python integration/scripts/build_knowledge_units.py --preflight --inventory $inventory --model gpt-5.6-luna --report $preflightPath
$preflight = Get-Content -LiteralPath $preflightPath -Raw | ConvertFrom-Json
if ($preflight.status -ne 'ready' -or $preflight.pilot_gate_status -ne 'PASS') { throw 'Full-run preflight not ready' }

$fullReportPath = 'integration/analysis/ai_context/knowledge_unit_full_run_report.json'
python integration/scripts/build_knowledge_units.py --start --inventory $inventory --model $preflight.actual_model_id --config-hash $preflight.config_hash --report $fullReportPath
$full = Get-Content -LiteralPath $fullReportPath -Raw | ConvertFrom-Json
$runId = [string]$full.run_id
if ([string]::IsNullOrWhiteSpace($runId)) { throw 'full report missing run_id' }
python integration/scripts/evaluate_knowledge_unit_extraction.py --run-id $runId --report $fullReportPath --strict
$full = Get-Content -LiteralPath $fullReportPath -Raw | ConvertFrom-Json
if ($full.gate.status -ne 'PASS' -or $full.inventory_hash -ne $preflight.inventory_hash -or $full.actual_model_id -ne $preflight.actual_model_id) { throw 'Full extraction gate or lineage failed' }

$canonicalReportPath = 'integration/analysis/ai_context/knowledge_unit_canonical_report.json'
python integration/scripts/build_canonical_knowledge_units.py --extraction-run-id $runId --write --report $canonicalReportPath
$canonical = Get-Content -LiteralPath $canonicalReportPath -Raw | ConvertFrom-Json
if ($canonical.gate.status -ne 'PASS' -or $canonical.extraction_run_id -ne $runId) { throw 'Canonical gate or lineage failed' }

$candidateBuildPath = 'integration/analysis/ai_context/knowledge_unit_candidate_build.json'
python integration/scripts/build_knowledge_unit_vector_store.py --canonical-build-id $canonical.canonical_build_id --write --report $candidateBuildPath
$candidate = Get-Content -LiteralPath $candidateBuildPath -Raw | ConvertFrom-Json
$candidateCollection = [string]$candidate.collection_name
if ([string]::IsNullOrWhiteSpace($candidateCollection) -or $candidate.reconcile.status -ne 'PASS') { throw 'Candidate build/reconcile failed' }

$abPath = 'integration/analysis/ai_context/knowledge_unit_candidate_ab.json'
python integration/scripts/evaluate_knowledge_unit_rag.py --dataset frozen-test --candidate $candidateCollection --report $abPath
$ab = Get-Content -LiteralPath $abPath -Raw | ConvertFrom-Json
if ($ab.gate.status -ne 'PASS' -or $ab.candidate_checksum -ne $candidate.collection_checksum -or $ab.canonical_build_id -ne $canonical.canonical_build_id) { throw 'Frozen A/B gate or lineage failed' }
```

`knowledge_unit_full_run_report.json` is the sole source of `run_id`; `knowledge_unit_canonical_report.json` supplies `canonical_build_id`; `knowledge_unit_candidate_build.json` supplies `collection_name`, `index_build_id`, and `collection_checksum`. Gate reports must expose `gate.status`, per-check results, and nonzero process exit on failure. Required lineage equality is `preflight.inventory_hash == full.inventory_hash`, `full.run_id == canonical.extraction_run_id`, `canonical.canonical_build_id == candidate.canonical_build_id == A/B.canonical_build_id`, and `candidate.collection_checksum == A/B.candidate_checksum`.
